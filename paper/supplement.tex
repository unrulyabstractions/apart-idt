% AAAI forbids hyperref, so \autoref does not exist here. The fragments use it
% with a typed label prefix, and the type is recoverable from that prefix, so
% the reference reads as it did in the working paper. Mapping every label to
% one word instead would have broken twenty-four table references.
\makeatletter
\newcommand{\autoref}[1]{%
  \IfBeginWith{#1}{tab:}{Table~\ref{#1}}{%
  \IfBeginWith{#1}{fig:}{Figure~\ref{#1}}{%
  \IfBeginWith{#1}{eq:}{Equation~\ref{#1}}{%
  \IfBeginWith{#1}{app:}{Appendix~\ref{#1}}{%
  \IfBeginWith{#1}{sec:}{Section~\ref{#1}}{\ref{#1}}}}}}}
\makeatother
