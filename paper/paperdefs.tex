% Floats are barriered at each \section (the placeins option above), so a table
% can never drift out of the section that discusses it. They are deliberately
% NOT barriered at each subsection: that flushes floats at every subsection
% head, and with \raggedbottom the slack collects as half a blank page.
%
% The default float separations leave a page of white space when several small
% tables sit in a row, so they are tightened here.
\setlength{\intextsep}{9pt plus 2pt minus 2pt}
\setlength{\textfloatsep}{11pt plus 2pt minus 2pt}
\setlength{\floatsep}{9pt plus 2pt minus 2pt}

% arxiv.sty zeroes abovecaptionskip on the assumption captions sit above tables.
% Every caption here sits below one, so the bottom rule and the caption touch.
\AtBeginEnvironment{table}{\setlength{\abovecaptionskip}{8pt}%
  \setlength{\belowcaptionskip}{0pt}}
\AtBeginEnvironment{longtable}{\setlength{\abovecaptionskip}{8pt}%
  \setlength{\belowcaptionskip}{0pt}}
